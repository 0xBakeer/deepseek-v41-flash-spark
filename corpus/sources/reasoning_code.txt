from pathlib import Path

def log_bytes(root) -> int:
    total = 0
    for p in Path(root).rglob("*.log"):
        try:
            if p.is_file():
                total += p.stat().st_size
        except OSError:

=== PROMPT
Print groups of duplicate files under a tree, matched by content.
=== THINK
Hashing every file is wasteful -- group by `st_size` first, then hash only the size collisions. Most trees have almost none, so that's the whole I/O win.

For the digest: `hashlib.file_digest(f, "sha256")` since 3.11 does the chunked read itself, no hand-rolled 64 KB loop.

```
by_size = defaultdict(list)
for p in Path(root).rglob("*"):
    if p.is_file():
        by_size[p.stat().st_size].append(p)
```
`is_file()` follows symlinks, so a link and its target land in the same group and report as dupes. Add `and not p.is_symlink()`.

Second pass as a dict comprehension? No -- appending into groups is a statement and I want a `try` around the open anyway. Loop.

And skip size 0: every empty file matches every other one, pure noise.

Settled: size buckets first, hash only the collisions, empties dropped. Writing it now.
=== ANSWER
```python
import hashlib
from collections import defaultdict
from pathlib import Path

def duplicates(root):
    by_size = defaultdict(list)
    for p in Path(root).rglob("*"):
        if p.is_file() and not p.is_symlink():
            try:
                size = p.stat().st_size
            except OSError:
                continue
            if size:
                by_size[size].append(p)

    by_hash = defaultdict(list)
    for group in by_size.values():
        if len(group) < 2:
            continue
        for p in group:
            try:
                with p.open("rb") as f:
                    by_hash[hashlib.file_digest(f, "sha256").hexdigest()].append(p)
            except OSError:
                continue
    return [g for g in by_hash.values() if len(g) > 1]
```
